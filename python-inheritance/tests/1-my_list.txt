>>> import importlib
>>> mod = importlib.import_module("1-my_list")
>>> MyList = mod.MyList

>>> MyList([2,1,4,4]).print_sorted()
[1, 2, 4, 4]
>>> MyList([1,1,1,1]).print_sorted()
[1, 1, 1, 1]
>>> MyList([]).print_sorted()
[]
>>> MyList([1,1,-1,1]).print_sorted()
[-1, 1, 1, 1]
>>> MyList([1,1,1,1]).append(-1)
None
>>> print(MyList([1,1,1,1]))
[1, 1, 1, 1]
